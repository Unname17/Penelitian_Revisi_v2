def safe_translate(text, max_retries=3):
    # Bing cukup kuat menampung hingga 1000 karakter per request
    chunks = textwrap.wrap(text, width=1000, break_long_words=False)
    translated_chunks = []

    for chunk in chunks:
        success = False
        wait_time = 3 

        for attempt in range(max_retries):
            try:
                # Jeda acak 2.5 hingga 4.5 detik agar terlihat seperti manusia
                time.sleep(random.uniform(2.5, 4.5)) 
                
                # MENGGUNAKAN MESIN BING
                res = ts.translate_text(chunk, translator='bing', from_language='id', to_language='en')
                
                translated_chunks.append(res)
                success = True
                break
            except Exception as e:
                print(f"    -> Error Bing: {e} | Menunggu {wait_time} detik... (Percobaan {attempt+1}/{max_retries})")
                time.sleep(wait_time)
                wait_time *= 2 

        if not success:
            return None

    return " ".join(translated_chunks)
