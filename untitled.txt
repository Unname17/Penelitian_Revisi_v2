# --- Tahap 1: Cleaning (Cell 15) ---
def clean_text(text):
    if text is None or not isinstance(text, str):
        return ""
    text = re.sub(r'\n+', ' ', text)
    text = re.sub(r'https?://\S+|www\.\S+', '', text)
    text = re.sub(r'@[^\s]+', '', text)
    text = re.sub(r'\brt\b', '', text, flags=re.IGNORECASE)
    text = re.sub(r'<.*?>', '', text)
    emoji_pattern = re.compile("["
        u"\U0001F600-\U0001F64F" u"\U0001F300-\U0001F5FF" u"\U0001F680-\U0001F6FF"
        u"\U0001F700-\U0001F77F" u"\U0001F780-\U0001F7FF" u"\U0001F800-\U0001F8FF"
        u"\U0001F900-\U0001F9FF" u"\U0001FA00-\U0001FA6F" u"\U0001FA70-\U0001FAFF"
        u"\U0001F004-\U0001F0CF" "]+", flags=re.UNICODE)
    text = emoji_pattern.sub(r'', text)
    text = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)
    text = re.sub(r'\b\d+\b', '', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

# --- Tahap 2: Normalisasi kata tidak baku (Cell 19) ---
def normalisasi_baku(text, kamus_tidak_baku):
    if not isinstance(text, str) or not text.strip():
        return ""
    tokens = text.split()
    blacklist_nama = {'mutia', 'prof', 'harjo', 'dekan', 'mbak'}
    replaced_words = []
    for token in tokens:
        token_lower = token.lower()
        if len(token_lower) <= 1 or token_lower.isdigit():
            continue
        if token_lower in blacklist_nama:
            continue
        if token_lower in kamus_tidak_baku:
            baku_word = kamus_tidak_baku[token_lower]
            replaced_words.append(baku_word if isinstance(baku_word, str) else token)
        else:
            replaced_words.append(token)
    return ' '.join(replaced_words)

# --- Tahap 3: Pre-translate normalization (Cell 21, Lapis 1) ---
def pre_translate_normalization(text):
    if not isinstance(text, str):
        return ""
    text = re.sub(r'\bs1s2\b', 'bachelor and master degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs2s3\b', 'master and doctoral degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs1\b', 'bachelor degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs2\b', 'master degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs3\b', 'doctoral degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bsd\b', 'institution', text, flags=re.IGNORECASE)
    return text

# --- Tahap 4: Post-translate cleaning (Cell 21, Lapis 2) ---
def post_translate_cleaning(text):
    if not isinstance(text, str):
        return ""
    text = re.sub(r'\bdosen\b', 'lecturer', text, flags=re.IGNORECASE)
    text = re.sub(r'\bgaji\b', 'salary', text, flags=re.IGNORECASE)
    text = re.sub(r'\bahli\b', 'expert', text, flags=re.IGNORECASE)
    text = re.sub(r'\bbayar\b', 'salary', text, flags=re.IGNORECASE)
    text = re.sub(r'\bdpr\b', 'government', text, flags=re.IGNORECASE)
    text = re.sub(r'([^0-9A-Za-z \t])|(\w+:\/\/\S+)', ' ', text)
    text = text.lower()
    text = re.sub(r'\b s \b', ' ', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def apply_synonyms(text, synonym_dict):
    if not isinstance(text, str) or not text.strip():
        return ""
    text = re.sub(r'\bcivil servant\b', 'status', text, flags=re.IGNORECASE)
    text = re.sub(r'\bcivil servants\b', 'status', text, flags=re.IGNORECASE)
    text = re.sub(r'\bnon asn\b', 'status', text, flags=re.IGNORECASE)
    words = text.split()
    replaced = [synonym_dict[w.lower()] if w.lower() in synonym_dict else w for w in words]
    return ' '.join(replaced)

    'no', 'not', 'nor', 'neither', 'never', 'cannot', 'cant', 'dont', 'doesnt',

all_stopwords = stop_words_id.union(custom_blacklist).union(stop_words_en)
final_blacklist = all_stopwords - negation_words

def remove_stopwords(text):
    if not isinstance(text, str):
        return ""
    words = text.split()
    filtered = [w.lower() for w in words if len(w.lower()) > 1 and not w.lower().isdigit() and w.lower() not in final_blacklist]
    return " ".join(filtered)

def clean_text(text):
    if text is None or not isinstance(text, str): return ""
    text = re.sub(r'\n+', ' ', text)
    text = re.sub(r'https?://\S+|www\.\S+', '', text)
    text = re.sub(r'@[^\s]+', '', text)
    text = re.sub(r'\brt\b', '', text, flags=re.IGNORECASE)
    text = re.sub(r'<.*?>', '', text)
    emoji_pattern = re.compile("["
        u"\U0001F600-\U0001F64F" u"\U0001F300-\U0001F5FF" u"\U0001F680-\U0001F6FF"
        u"\U0001F700-\U0001F77F" u"\U0001F780-\U0001F7FF" u"\U0001F800-\U0001F8FF"
        u"\U0001F900-\U0001F9FF" u"\U0001FA00-\U0001FA6F" u"\U0001FA70-\U0001FAFF"
        u"\U0001F004-\U0001F0CF" "]+", flags=re.UNICODE)
    text = emoji_pattern.sub(r'', text)
    text = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)
    text = re.sub(r'\b\d+\b', '', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def normalisasi_baku(text, kamus_tidak_baku):
    if not isinstance(text, str) or not text.strip(): return ""
    tokens = text.split()
    blacklist_nama = {'mutia', 'prof', 'harjo', 'dekan', 'mbak'}
    replaced_words = []
    for token in tokens:
        token_lower = token.lower()
        if len(token_lower) <= 1 or token_lower.isdigit(): continue
        if token_lower in blacklist_nama: continue
        if token_lower in kamus_tidak_baku:
            baku_word = kamus_tidak_baku[token_lower]
            replaced_words.append(baku_word if isinstance(baku_word, str) else token)
        else:
            replaced_words.append(token)
    return ' '.join(replaced_words)

def pre_translate_normalization(text):
    if not isinstance(text, str): return ""
    text = re.sub(r'\bs1s2\b', 'bachelor and master degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs2s3\b', 'master and doctoral degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs1\b', 'bachelor degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs2\b', 'master degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bs3\b', 'doctoral degree', text, flags=re.IGNORECASE)
    text = re.sub(r'\bsd\b', 'institution', text, flags=re.IGNORECASE)
    return text

def post_translate_cleaning(text):
    if not isinstance(text, str): return ""
    text = re.sub(r'\bdosen\b', 'lecturer', text, flags=re.IGNORECASE)
    text = re.sub(r'\bgaji\b', 'salary', text, flags=re.IGNORECASE)
    text = re.sub(r'\bahli\b', 'expert', text, flags=re.IGNORECASE)
    text = re.sub(r'\bbayar\b', 'salary', text, flags=re.IGNORECASE)
    text = re.sub(r'\bdpr\b', 'government', text, flags=re.IGNORECASE)
    text = re.sub(r'([^0-9A-Za-z \t])|(\w+:\/\/\S+)', ' ', text)
    text = text.lower()
    text = re.sub(r'\b s \b', ' ', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def apply_synonyms(text, synonym_dict):
    if not isinstance(text, str) or not text.strip(): return ""
    text = re.sub(r'\bcivil servant\b', 'status', text, flags=re.IGNORECASE)
    text = re.sub(r'\bcivil servants\b', 'status', text, flags=re.IGNORECASE)
    
    # Hapus synonym untuk 'non asn' di sini jika Anda ingin entitas 'non_asn' tertangkap LDA. 
    # Namun karena SVM butuh 'status', kita biarkan, TAPI kita akan ubah untuk LDA di fungsi khusus.
    
    words = text.split()
    replaced = [synonym_dict[w.lower()] if w.lower() in synonym_dict else w for w in words]
    return ' '.join(replaced)

def remove_stopwords(text):

    words = text.split()
    filtered = [w.lower() for w in words if len(w.lower()) > 1 and not w.lower().isdigit() and w.lower() not in final_blacklist]
    return " ".join(filtered)

                            lda_text = lda_text.replace("prabowo subianto", "prabowo_subianto")
                            lda_text = lda_text.replace("bpjs ketenagakerjaan", "bpjs_ketenagakerjaan")
                            lda_text = lda_text.replace("non asn", "non_asn")
                            
                            tokens_lda = [w for w in lda_text.split() if w.lower() not in custom_stopwords_lda]
                            bow = lda_dict.doc2bow(tokens_lda)
                            topics = lda_model.get_document_topics(bow)
                            aspek = "Tidak Terdeteksi"
                            if topics and len(topics) > 0:
                                best_topic = max(topics, key=lambda x: x[1])[0]
                                aspek = aspect_names.get(best_topic, "Lainnya")

                            # PIPELINE SVM
                            tfidf = tfidf_vectorizer.transform([final_text])
                            pred_sentimen = svm_model.predict(tfidf)[0]

                            hasil_topik.append(aspek)
                            hasil_sentimen.append(pred_sentimen)
                            
                            hasil_normalisasi.append(step2_normalisasi)
                            hasil_translate.append(translated_raw)
                            hasil_bersih.append(final_text)
                            hasil_lda.append(" ".join(tokens_lda))

                        except Exception as e:
                            hasil_topik.append("Error Processing")
                            hasil_sentimen.append("Error")
                            hasil_normalisasi.append("Error")
                            hasil_translate.append("Error")
                            hasil_bersih.append("Error")
                            hasil_lda.append("Error")

                        progress_bar.progress((i + 1) / total_data)
                        status_text.text(f"Memproses data {i + 1} dari {total_data}...")

                    status_text.success("✅ Pemrosesan Masal Selesai!")

                    # Assign to DataFrame
                    df_proses['Teks_Normalisasi'] = hasil_normalisasi
                    df_proses['Teks_Translate'] = hasil_translate
                    df_proses['Teks_Bersih'] = hasil_bersih
                    df_proses['Teks_LDA_Tokens'] = hasil_lda
                    df_proses['Prediksi_Topik'] = hasil_topik
                    df_proses['Prediksi_Sentimen'] = hasil_sentimen

                    df_hasil = df_proses[df_proses['Prediksi_Sentimen'] != 'Error'].copy()

                    st.markdown("---")
                    st.markdown("### 📊 Dashboard Rekomendasi Kebijakan")

                    total = len(df_hasil)
                    tot_pos = len(df_hasil[df_hasil['Prediksi_Sentimen'] == 'Positif'])
                    tot_neg = len(df_hasil[df_hasil['Prediksi_Sentimen'] == 'Negatif'])

                    m1, m2, m3 = st.columns(3)
                    m1.metric("Total Data Dianalisis", f"{total} Data")
                    m2.metric("Sentimen Positif", f"{tot_pos} Data")
                    m3.metric("Sentimen Negatif", f"{tot_neg} Data")

                    col_chart1, col_chart2 = st.columns([1, 1.5])
                    warna_sentimen = {'Positif': '#0ea5e9', 'Negatif': '#ef4444'}

                    with col_chart1:
                        st.markdown("**Distribusi Sentimen**")
                        if total > 0:
                            fig_pie = px.pie(df_hasil, names='Prediksi_Sentimen', hole=0.5, 
                                             color='Prediksi_Sentimen', color_discrete_map=warna_sentimen)
                            fig_pie.update_layout(margin=dict(t=20, b=20, l=20, r=20), showlegend=True)
                            st.plotly_chart(fig_pie, use_container_width=True)
                        else:
                            st.info("Tidak ada data untuk ditampilkan.")

                    with col_chart2:
                        st.markdown("**Prioritas Isu Kebijakan (Topik vs Sentimen)**")
                        if total > 0:
                            df_group = df_hasil.groupby(['Prediksi_Topik', 'Prediksi_Sentimen']).size().reset_index(name='Jumlah')
                            fig_bar = px.bar(df_group, x='Prediksi_Topik', y='Jumlah', color='Prediksi_Sentimen',
                                             color_discrete_map=warna_sentimen, barmode='stack', text='Jumlah')
                            fig_bar.update_layout(xaxis_title="Topik (Aspek)", yaxis_title="Jumlah Aspirasi",
                                                  xaxis={'categoryorder':'total descending'}, 
                                                  margin=dict(t=20, b=20, l=20, r=20))
                            st.plotly_chart(fig_bar, use_container_width=True)

                    if sentiment_col_actual:
                        st.markdown("---")
                        st.markdown("### 🎯 Evaluasi Model Klasifikasi")

                        y_true = df_hasil[sentiment_col_actual].astype(str).str.strip().str.title()
                        y_pred = df_hasil['Prediksi_Sentimen'].astype(str).str.strip().str.title()

                        acc = accuracy_score(y_true, y_pred)
                        prec = precision_score(y_true, y_pred, average='weighted', zero_division=0)
                        rec = recall_score(y_true, y_pred, average='weighted', zero_division=0)
                        f1 = f1_score(y_true, y_pred, average='weighted', zero_division=0)

                        st.markdown("#### Performa Model")
                        met1, met2, met3, met4 = st.columns(4)
                        met1.metric("Accuracy", f"{acc:.4f}")
                        met2.metric("Precision (Weighted)", f"{prec:.4f}")
                        met3.metric("Recall (Weighted)", f"{rec:.4f}")
                        met4.metric("F1-Score (Weighted)", f"{f1:.4f}")

                        st.markdown("<br>", unsafe_allow_html=True)
                        st.markdown("#### Confusion Matrix")

                        labels_cm = ['Negatif', 'Positif']
                        cm = confusion_matrix(y_true, y_pred, labels=labels_cm)

                        fig_cm = px.imshow(cm, text_auto=True, 
                                           x=['Prediksi Negatif', 'Prediksi Positif'], 
                                           y=['Aktual Negatif', 'Aktual Positif'],
                                           color_continuous_scale='Oranges',
                                           aspect="auto")
                        fig_cm.update_layout(margin=dict(t=30, b=30, l=30, r=30))

                        st.plotly_chart(fig_cm, use_container_width=True)

                    st.markdown("### 📋 Tabel Data Hasil Analisis & Preprocessing")

                    kolom_tampil = [kolom_teks, 'Teks_Normalisasi', 'Teks_Translate', 'Teks_Bersih', 'Teks_LDA_Tokens', 'Prediksi_Topik', 'Prediksi_Sentimen']
